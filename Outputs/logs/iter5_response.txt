import pandas as pd
import numpy as np
from sklearn.ensemble import HistGradientBoostingClassifier, RandomForestClassifier
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from sklearn.base import BaseEstimator, ClassifierMixin

def engineer_features(df: pd.DataFrame) -> pd.DataFrame:
    df_feat = df.copy()
    
    # Drop Day and Signal if present
    drop_cols = ['Day', 'Signal']
    feature_cols = [c for c in df_feat.columns if c not in drop_cols]
    
    X = df_feat[feature_cols].copy()
    
    # Microstructure Feature Engineering
    X['NOF_Std'] = np.sqrt(np.maximum(X['NOF_Variance'], 1e-12))
    X['Return_Std'] = np.sqrt(np.maximum(X['Return_Variance'], 1e-12))
    
    # Order Flow Signal-to-Noise Ratio (Z-Score)
    X['NOF_SNR'] = X['Mean_NetOrderFlow'] / (X['NOF_Std'] + 1e-8)
    
    # Kyle's Lambda Price Impact Proxy
    X['Price_Impact_Lambda'] = X['Price_Change'] / (X['Mean_NetOrderFlow'].abs() + 1e-8)
    
    # Return Sharpe / Risk-Adjusted Return
    X['Return_Sharpe'] = X['Daily_Return'] / (X['Return_Std'] + 1e-8)
    
    # Non-linear Tail and Skewness Interactions
    X['Flow_Skew_Interaction'] = X['Mean_NetOrderFlow'] * X['Return_Skewness']
    X['Flow_Kurt_Interaction'] = X['Mean_NetOrderFlow'] * X['Return_Kurtosis']
    X['Tail_Intensity'] = X['Return_Kurtosis'] * X['Return_Variance']
    
    return X

class FinalEnsembleClassifier(BaseEstimator, ClassifierMixin):
    def __init__(self):
        self.scaler = StandardScaler()
        self.gbc = HistGradientBoostingClassifier(
            max_iter=200,
            max_depth=3,
            learning_rate=0.03,
            l2_regularization=2.0,
            random_state=42
        )
        self.rfc = RandomForestClassifier(
            n_estimators=150,
            max_depth=5,
            min_samples_split=15,
            random_state=42,
            n_jobs=-1
        )
        self.lr = LogisticRegression(C=0.3, penalty='l2', random_state=42)
        
    def fit(self, X, y):
        X_scaled = self.scaler.fit_transform(X)
        self.gbc.fit(X_scaled, y)
        self.rfc.fit(X_scaled, y)
        self.lr.fit(X_scaled, y)
        return self
        
    def predict_proba(self, X):
        X_scaled = self.scaler.transform(X)
        p_gbc = self.gbc.predict_proba(X_scaled)
        p_rfc = self.rfc.predict_proba(X_scaled)
        p_lr = self.lr.predict_proba(X_scaled)
        # Weighted soft voting ensemble
        return 0.5 * p_gbc + 0.3 * p_rfc + 0.2 * p_lr

# Global model container
model_pipeline = None

def train_pred(df: pd.DataFrame) -> pd.DataFrame:
    global model_pipeline
    
    X_train = engineer_features(df)
    y_train = df['Signal'].astype(int)
    
    model_pipeline = FinalEnsembleClassifier()
    model_pipeline.fit(X_train, y_train)
    
    # Probabilities and empirical threshold calibration matching training base rate
    train_probs = model_pipeline.predict_proba(X_train)[:, 1]
    target_pos_rate = y_train.mean()
    sorted_probs = np.sort(train_probs)
    threshold_idx = int(len(sorted_probs) * (1 - target_pos_rate))
    optimal_threshold = sorted_probs[max(0, min(threshold_idx, len(sorted_probs)-1))]
    optimal_threshold = np.clip(optimal_threshold, 0.36, 0.44)
    
    model_pipeline.threshold_ = optimal_threshold
    
    train_preds = (train_probs >= optimal_threshold).astype(int)
    return pd.DataFrame({'Signal': train_preds}, index=df.index)

def test_pred(df: pd.DataFrame) -> pd.DataFrame:
    global model_pipeline
    
    X_test = engineer_features(df)
    test_probs = model_pipeline.predict_proba(X_test)[:, 1]
    
    threshold = getattr(model_pipeline, 'threshold_', 0.40)
    test_preds = (test_probs >= threshold).astype(int)
    
    return pd.DataFrame({'Signal': test_preds}, index=df.index)
